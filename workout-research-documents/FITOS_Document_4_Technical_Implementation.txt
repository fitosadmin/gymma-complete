2.2 STATE MANAGEMENT: RIVERPOD (Mandatory)
------------------------------------------
Every feature uses Riverpod for unidirectional data flow.

PATTERN:
- @riverpod async functions for API calls
- StateNotifier for complex form state (Assessment)
- FutureProvider for read-only data (Exercise library)

EXAMPLE ASSESSMENT STATE:
```dart
@freezed
class AssessmentState with _$AssessmentState {
  const factory AssessmentState({
    @Default({}) Map<String, dynamic> responses,
    @Default(0) int currentSectionIndex,
    @Default(false) bool isSubmitting,
    UserFitnessProfileVector? computedVector,
    Failure? error,
  }) = _AssessmentState;
}

@riverpod
class AssessmentController extends _$AssessmentController {
  @override
  AssessmentState build() => const AssessmentState();

  void updateResponse(String fieldId, dynamic value) {
    state = state.copyWith(
      responses: {...state.responses, fieldId: value},
    );
    _reevaluateBranching();
  }

  void _reevaluateBranching() {
    // Document 1: Dynamic form behavior
    final expScore = _computeExperienceScore(state.responses);
    if (expScore < 20) {
      state = state.copyWith(
        hiddenSections: {'S7', 'S8'},
        forcedSplit: SplitType.fullBody,
      );
    }
  }
}
```

REAL-TIME COMPUTATION:
```dart
double computeRecoveryCapacity(Map<String, dynamic> r) {
  final sleep = (r['S2_SLEEP'] as double) * (r['S2_SLEEP_QUALITY'] as int) * 4;
  final occupation = _occupationMap[r['S2_OCCUPATION']]! * 15;
  final stress = (4 - _stressMap[r['S2_STRESS']]!) * 20;
  final nutrition = _nutritionMap[r['S2_NUTRITION']]! * 25;
  final alcohol = max(0, 21 - (r['S2_ALCOHOL'] as int)) * 1.5;
  final raw = sleep + occupation + stress + nutrition + alcohol;
  return ((raw / 491.5) * 100).clamp(0.0, 100.0);
}
```

2.4 DYNAMIC FORM NAVIGATION
---------------------------
Use GoRouter with state guards.

```dart
final router = GoRouter(
  routes: [
    GoRoute(
      path: '/assessment/:sectionId',
      builder: (context, state) {
        final sectionId = state.pathParameters['sectionId']!;
        final assessment = ref.watch(assessmentControllerProvider);

        // Guard: Cannot skip ahead
        final allowedIndex = _getMaxAllowedIndex(assessment);
        final requestedIndex = _sectionIndexMap[sectionId]!;

        if (requestedIndex > allowedIndex) {
          return const RedirectScreen(to: '/assessment/S0');
        }

        // Guard: Section 7/8 visibility
        if ((sectionId == 'S7' && assessment.hiddenSections.contains('S7')) ||
            (sectionId == 'S8' && assessment.hiddenSections.contains('S8'))) {
          return const RedirectScreen(to: '/assessment/S4');
        }

        return AssessmentSectionScreen(sectionId: sectionId);
      },
    ),
  ],
);
```

```typescript
@Injectable()
export class RecommendationEngineService {
  constructor(
    private prisma: PrismaService,
    private exerciseMatcher: ExerciseMatcherService,
  ) {}

  async generatePlan(assessmentId: string): Promise<WorkoutPlan> {
    // STEP 1: INGEST USER VECTOR U
    const assessment = await this.prisma.assessment.findUnique({
      where: { id: assessmentId },
    });
    const U = assessment.computedVector as UserFitnessProfileVector;

    // STEP 2: LOAD GYM EXERCISE DATABASE
    const gymExercises = await this.prisma.gymExerciseDatabase.findMany({
      where: { gymId: assessment.user.gymId, isAvailable: true },
      include: { exercise: true },
    });

    // STEP 3: APPLY HARD FILTERS
    const candidates = gymExercises.filter(({ exercise }) => {
      if (exercise.experienceMinimum > U.Experience_Score) return false;
      // Equipment gate handled by gym_exercise_database
      // Blacklist gate
      const blacklist = assessment.responses['S8_DISLIKES'] || [];
      if (blacklist.includes(exercise.code)) return false;
      // Contraindication gate
      // ... (prisma query for contraindications)
      // Mobility gate
      if (exercise.mobilityRequired > U.Mobility_Score) return false;
      return true;
    }).map(ge => ge.exercise);

    // STEP 4: COMPUTE MATCH SCORES
    const scoredExercises = candidates.map(e => ({
      exercise: e,
      matchScore: this.exerciseMatcher.computeMatchScore(e, U, assessment.responses),
    })).sort((a, b) => b.matchScore - a.matchScore);

    // STEP 5: DETERMINE PROGRAM PARAMETERS
    const F = this.calculateFrequency(U.Availability_Score, U.Recovery_Capacity, assessment.responses['S5_DAYS']);
    const V = this.calculateVolume(U.Experience_Score, U.Recovery_Capacity);
    const intensityRange = this.calculateIntensity(assessment.responses['S4_PRIMARY'], U.Experience_Score);
    const split = this.selectSplit(F, U.Experience_Score, assessment.responses['S8_SPLIT']);

    // STEP 6-7: ALLOCATE VOLUME & SELECT EXERCISES
    const sessions = this.buildSessions(scoredExercises, F, V, split, U, assessment.responses);

    // STEP 8: ASSIGN SETS, REPS, REST, RPE
    const periodizedSessions = this.applyPeriodization(sessions, U.Experience_Score, assessment.responses);

    // STEP 9: OUTPUT & STORE
    return this.prisma.workoutPlan.create({
      data: {
        assessmentId,
        userId: assessment.userId,
        metadata: { version: '1.0.0', generatedAt: new Date().toISOString() },
        programParameters: { frequency: F, split, volume: V, intensityModel: intensityRange },
        sessions: periodizedSessions as any,
        progressionRules: this.getProgressionRules(U.Experience_Score),
        safetyFlags: this.generateSafetyFlags(U),
      },
    });
  }

  private computeMatchScore(
    exercise: Exercise,
    U: UserFitnessProfileVector,
    responses: any,
  ): number {
    const Ge = this.getGoalCompatibility(exercise, responses['S4_PRIMARY'], responses['S4_SECONDARY']);
    const difficultyFit = 100 * Math.exp(
      -Math.pow(exercise.difficultyScore - (U.Experience_Score - 10), 2) / (2 * Math.pow(20, 2))
    );
    const mobilityFit = 100 - Math.abs(exercise.mobilityRequired - U.Mobility_Score);
    const safetyMargin = 100 - exercise.injuryRiskFactor;
    const equipmentMatch = 100; // Already filtered
    const musclePriority = this.computeMusclePriority(exercise, responses['S4_TARGET_AREAS'] || []);

    return (
      0.30 * Ge +
      0.25 * difficultyFit +
      0.10 * mobilityFit +
      0.15 * safetyMargin +
      0.10 * equipmentMatch +
      0.10 * musclePriority
    );
  }

  private calculateFrequency(A: number, R: number, requestedDays: number): number {
    const maxFreq = Math.floor((A * 7) / 100);
    const recoveryLimit = R < 30 ? 2 : R < 50 ? 3 : R < 70 ? 4 : R < 85 ? 5 : 6;
    return Math.min(maxFreq, recoveryLimit, requestedDays);
  }

  private calculateVolume(E: number, R: number): number {
    let base = 10;
    if (E >= 20 && E < 40) base = 12;
    else if (E >= 40 && E < 60) base = 15;
    else if (E >= 60 && E <= 80) base = 18;
    else if (E > 80) base = 20;
    return base * (R / 100);
  }

  private selectSplit(F: number, E: number, preferredSplit?: string): string {
    const matrix = {
      1: { any: 'Full Body' },
      2: { low: 'Full Body', high: 'Upper/Lower' },
      3: { low: 'Full Body', high: 'Push/Pull/Legs' },
      4: { low: 'Upper/Lower', high: 'Upper/Lower' },
      5: { low: 'Upper/Lower', high: 'Push/Pull/Legs' },
      6: { low: 'Push/Pull/Legs', high: 'Bro Split / Custom' },
    };
    const expTier = E < 20 ? 'low' : E < 60 ? 'mid' : 'high';
    // ... matrix lookup logic with preference validation
    return 'Full Body'; // Simplified
  }
}
```

```typescript
@Injectable()
export class VectorComputationService {
  computeUserVector(responses: AssessmentResponses): UserFitnessProfileVector {
    return {
      Goal_Alignment: this.computeG(responses),
      Recovery_Capacity: this.computeR(responses),
      Mobility_Score: this.computeM(responses),
      Strength_Level: this.computeS(responses),
      Experience_Score: this.computeE(responses),
      Availability_Score: this.computeA(responses),
      Injury_Risk: this.computeI(responses),
      Preference_Alignment: 50, // Computed post-matching
    };
  }

  private computeG(r: AssessmentResponses): number {
    const primary = r.S4_PRIMARY;
    const secondary = r.S4_SECONDARY;
    if (!secondary) return 100;

    const penaltyMatrix: Record<string, Record<string, number>> = {
      'Strength': { 'Hypertrophy': 0.3, 'Endurance': 0.8, 'Power': 0.1, 'General_Health': 0.2, 'Recomposition': 0.3 },
      'Hypertrophy': { 'Strength': 0.3, 'Endurance': 0.6, 'Power': 0.4, 'General_Health': 0.2, 'Recomposition': 0.1 },
      'Endurance': { 'Strength': 0.8, 'Hypertrophy': 0.6, 'Power': 0.7, 'General_Health': 0.3, 'Recomposition': 0.5 },
      'Power': { 'Strength': 0.1, 'Hypertrophy': 0.4, 'Endurance': 0.7, 'General_Health': 0.3, 'Recomposition': 0.4 },
      'General_Health': { 'Strength': 0.2, 'Hypertrophy': 0.2, 'Endurance': 0.3, 'Power': 0.3, 'Recomposition': 0.1 },
      'Recomposition': { 'Strength': 0.3, 'Hypertrophy': 0.1, 'Endurance': 0.5, 'Power': 0.4, 'General_Health': 0.1 },
    };

    const penalty = penaltyMatrix[primary]?.[secondary] ?? 0;
    return Math.max(0, 100 - (penalty * 15));
  }

  private computeR(r: AssessmentResponses): number {
    const occupationMap = { 'Sedentary': 1, 'Light': 2, 'Active': 3, 'Very_Active': 4 };
    const stressMap = { 'Low': 1, 'Med': 2, 'High': 3 };
    const nutritionMap = { 'Poor': 1, 'Inconsistent': 2, 'Consistent': 3, 'Strict': 4 };

    const raw = (
      (r.S2_SLEEP * r.S2_SLEEP_QUALITY * 4) +
      (occupationMap[r.S2_OCCUPATION] * 15) +
      ((4 - stressMap[r.S2_STRESS]) * 20) +
      (nutritionMap[r.S2_NUTRITION] * 25) +
      (Math.max(0, 21 - (r.S2_ALCOHOL || 0)) * 1.5)
    );
    return Math.min(100, (raw / 491.5) * 100);
  }

  private computeE(r: AssessmentResponses): number {
    const durationMap = { 'Never': 0, '<6mo': 15, '6-12mo': 35, '1-2yr': 55, '2-5yr': 80, '5+yr': 100 };
    const durationPoints = durationMap[r.S3_DURATION];
    const compoundPoints = (r.S3_COMPOUNDS?.length || 0) * 10;
    const programPoints = r.S3_PROGRAM ? 15 : 0;
    const freqMap = [0, 5, 5, 10, 10, 15, 15, 20, 20, 20, 20, 20, 20, 20, 20, 20, 20, 20, 20, 20, 20, 20];
    const frequencyPoints = freqMap[Math.min(r.S3_FREQUENCY, 21)] || 0;

    const raw = (
      durationPoints * 0.50 +
      compoundPoints * 0.30 +
      programPoints * 0.15 +
      frequencyPoints * 0.05
    );
    return Math.min(100, raw);
  }

  private computeA(r: AssessmentResponses): number {
    const gymFactor = { 'Home': 8, 'Commercial': 10, 'Outdoor': 7 };
    const raw = (r.S5_DAYS * 12) + (r.S5_TIME * 0.5) + gymFactor[r.S5_GYM_TYPE];
    return Math.min(100, (raw / 154) * 100);
  }

  private computeI(r: AssessmentResponses): number {
    let base = 0;
    if (r.S0_Q2 || r.S0_Q3) base += 30;
    if (r.S0_Q4) base += 20;
    if (r.S0_Q5) base += 10;
    if (r.S6_CURRENT_INJURY) base += (r.S6_PAIN_SCALE || 0) * 5;
    if (r.S6_PAST_SURGERY) base += 15;
    if (r.S6_PHYSIO) base += 10;
    return Math.min(base, 100);
  }

  private computeM(r: AssessmentResponses): number {
    const E = this.computeE(r);
    if (E < 20) return 50;

    const squat = r.S7_SQUAT_DEPTH === 'Yes' ? 25 : r.S7_SQUAT_DEPTH === 'Only with raised heels' ? 15 : 5;
    const overhead = r.S7_OVERHEAD === 'Yes' ? 25 : r.S7_OVERHEAD === 'Partial' ? 15 : 5;
    const hinge = r.S7_HINGE === 'Yes' ? 20 : r.S7_HINGE === 'Almost' ? 12 : 5;
    const lunge = r.S7_LUNGE === 'Yes' ? 20 : r.S7_LUNGE === 'Unstable' ? 10 : 5;
    const rotary = r.S7_ROTARY === 'Yes' ? 10 : 5;

    return Math.min(100, squat + overhead + hinge + lunge + rotary);
  }

  private computeS(r: AssessmentResponses): number {
    const E = this.computeE(r);
    if (!r.S3_1RM_AWARE || E < 20) {
      return E * 0.8;
    }

    // Calculate relative strength ratios and compare to standards
    const squatRatio = r.S3_SQUAT_1RM / r.S1_WEIGHT;
    const deadliftRatio = r.S3_DEADLIFT_1RM / r.S1_WEIGHT;
    const benchRatio = r.S3_BENCH_1RM / r.S1_WEIGHT;

    // Sex-specific standards (simplified)
    const isFemale = r.S1_SEX === 'F';
    const squatStd = isFemale ? [0.65, 1.0, 1.3, 1.6] : [1.0, 1.5, 2.0, 2.5];
    const deadliftStd = isFemale ? [0.8, 1.2, 1.6, 2.0] : [1.2, 1.8, 2.5, 3.0];
    const benchStd = isFemale ? [0.4, 0.75, 1.1, 1.4] : [0.75, 1.25, 1.75, 2.25];

    const getPercentile = (ratio: number, standards: number[]) => {
      if (ratio < standards[0]) return ratio / standards[0] * 25;
      if (ratio < standards[1]) return 25 + (ratio - standards[0]) / (standards[1] - standards[0]) * 25;
      if (ratio < standards[2]) return 50 + (ratio - standards[1]) / (standards[2] - standards[1]) * 25;
      if (ratio < standards[3]) return 75 + (ratio - standards[2]) / (standards[3] - standards[2]) * 25;
      return Math.min(100, 100 + (ratio - standards[3]) / standards[3] * 25);
    };

    const avg = (getPercentile(squatRatio, squatStd) + 
                 getPercentile(deadliftRatio, deadliftStd) + 
                 getPercentile(benchRatio, benchStd)) / 3;
    return Math.min(100, avg);
  }
}
```

```typescript
@Injectable()
export class EdgeCaseService {
  handleNoExercises(candidates: Exercise[], userVector: UserFitnessProfileVector): Exercise[] {
    // 6.1: No available exercises after filtering
    if (candidates.length === 0) {
      // Fallback to bodyweight
      const bwFallback = await this.prisma.exercise.findMany({
        where: { equipment: { some: { equipment: { code: 'BW' } } } },
      });
      if (bwFallback.length > 0) return bwFallback;

      // Mobility-only plan
      throw new MobilityOnlyPlanException(
        'Your current constraints limit resistance training. Focus on movement quality.'
      );
    }
    return candidates;
  }

  checkFeasibility(U: UserFitnessProfileVector, responses: any): FeasibilityResult {
    // 6.2: Conflicting goals and availability
    const feasibility = (U.Availability_Score * 0.4) + (U.Experience_Score * 0.3) + (U.Recovery_Capacity * 0.3);
    if (responses.S4_PRIMARY === 'Power' && feasibility < 40) {
      return {
        isFeasible: false,
        message: 'Power training requires 4-5 days/week and 60+ minutes/session.',
        recommendations: ['Adjust goal to General Health', 'Keep goal but accept minimal progress', 'Update availability'],
      };
    }
    return { isFeasible: true };
  }

  detectPlateau(logs: PerformanceLog[]): PlateauDetectionResult {
    // 6.3: Performance stall detection
    const grouped = _.groupBy(logs, 'exerciseId');
    const stalledExercises = Object.entries(grouped).filter(([exId, exerciseLogs]) => {
      if (exerciseLogs.length < 3) return false;
      const lastThree = exerciseLogs.slice(-3);
      // Check if load or reps increased in any of last 3 sessions
      return !lastThree.some((log, i) => {
        if (i === 0) return false;
        return log.sets.some((set: any, j: number) => 
          set.load > lastThree[i-1].sets[j]?.load || set.reps > lastThree[i-1].sets[j]?.reps
        );
      });
    });

    if (stalledExercises.length >= 3) {
      return { isPlateau: true, affectedExercises: stalledExercises.map(([id]) => id) };
    }
    return { isPlateau: false };
  }
}
```
